
"""
Alphaopics - Complete Python Translation
=========================================

This module translates all alphaopics R functions to Python using properly
formatted CSV data exports from the alphaopics R package.

Functions:
----------
- photonenergy(wavelength) - Calculate photon energy (Joules)
- vlambda(power, wavelength) - Photopic illuminance/luminance
- blambda(power, wavelength, quantity) - Biological light detection
- govardovskii(lmax, wavelength) - Visual pigment sensitivity template
- aopicspecies() - Get species information
- alphaopic(power, wavelength, opsin, lmax, pfilter) - Main calculator
- generateaopicactionspec(opsin, lmax, pfilter, range_wl) - Generate action spectrum
"""

# Get the data directory
DATA_DIR = os.path.join(os.path.dirname(__file__), 'data')

# ============================================================================
# CORE PHOTOMETRIC CONSTANTS
# ============================================================================

h = 6.62607015e-34  # Planck's constant (J·s)
c = 299792458       # Speed of light (m/s)
K_m = 683           # Luminous efficacy of radiation (lm/W) at 555 nm


# ============================================================================
# DATA LOADING (cached)
# ============================================================================

_data_cache = {}

def _load_csv(filename):
    """Load and cache CSV data from the data directory"""
    if filename not in _data_cache:
        filepath = os.path.join(DATA_DIR, filename)
        if os.path.exists(filepath):
            _data_cache[filename] = pd.read_csv(filepath)
        else:
            _data_cache[filename] = None
    return _data_cache[filename]

def photonenergy(wavelength):
    """
    Calculate photon energy from wavelength.
    
    Args:
        wavelength: Wavelength in nm (scalar or array-like)
    
    Returns:
        Photon energy in Joules
    
    Formula: E = hc / λ
    """
    wavelength = np.asarray(wavelength, dtype=float)
    wavelength_m = wavelength * 1e-9  # Convert nm to meters
    energy = (h * c) / wavelength_m
    return energy


def vlambda(power, wavelength):
    """
    Calculate photopic illuminance (lux) or luminance (cd/m²).
    
    Uses the CIE V(λ) photopic sensitivity function.
    Matches R behavior: only uses wavelengths in both power and V(λ) datasets.
    
    Args:
        power: Spectral irradiance in W/m²·nm or radiance in W/m²·sr·nm (array)
        wavelength: Array of wavelengths in nm
    
    Returns:
        Photopic illuminance (lux) or luminance (cd/m²)
    """
    power = np.asarray(power, dtype=float)
    wavelength = np.asarray(wavelength, dtype=float)
    
    # Load V(λ) data
    vlambda_df = _load_csv('VisualStandards_vlambda.csv')
    if vlambda_df is None or len(vlambda_df) == 0:
        return 0.0
    
    # Merge on wavelengths (only include overlapping wavelengths, like R does)
    # Create dataframes for merging
    spd_df = pd.DataFrame({
        'wavelen': wavelength.astype(int),
        'power': power
    })
    vlam_df = pd.DataFrame({
        'wavelen': vlambda_df['wavelen'].astype(int),
        'vlambda': vlambda_df['photopic'].values
    })
    
    # Inner merge keeps only overlapping wavelengths
    merged = pd.merge(spd_df, vlam_df, on='wavelen', how='inner')
    
    if len(merged) == 0:
        return 0.0
    
    # Calculate photopic: lux = K_m * Σ(power * V(λ))
    photopic = K_m * np.sum(merged['power'].values * merged['vlambda'].values)
    return photopic


def blambda(power, wavelength, quantity="luminous"):
    """
    Biological light detection - Ba(λ), Bp(λ), Be(λ).
    
    Args:
        power: Spectral irradiance in W/m²·nm
        wavelength: Array of wavelengths in nm
        quantity: "luminous", "radiant", or "photon"
    
    Returns:
        Dictionary with 'Ba', 'Bp', 'Be' keys
    """
    power = np.asarray(power, dtype=float)
    wavelength = np.asarray(wavelength, dtype=float)
    
    # Load biological light detection standards
    ba_data = _load_csv('VisualStandards_balambda.csv')
    bp_data = _load_csv('VisualStandards_bplambda.csv')
    be_data = _load_csv('VisualStandards_belambda.csv')
    
    # Create SPD dataframe for merging
    spd_df = pd.DataFrame({
        'wavelen': wavelength.astype(int),
        'power': power
    })
    
    if quantity == "luminous":
        # Apply Ba(λ) with kavD65 normalization
        ba = 0.0
        bp = 0.0
        be = 0.0
        
        if ba_data is not None and len(ba_data) > 0:
            ba_df = pd.DataFrame({
                'wavelen': ba_data['wavelen'].astype(int),
                'balambda': ba_data['balambda'].values
            })
            merged_ba = pd.merge(spd_df, ba_df, on='wavelen', how='inner')
            if len(merged_ba) > 0:
                ka_val = ba_data['kavD65'].iloc[0]  # Single constant value
                ba = np.sum(merged_ba['power'].values * merged_ba['balambda'].values) / ka_val
        
        if bp_data is not None and len(bp_data) > 0:
            bp_df = pd.DataFrame({
                'wavelen': bp_data['wavelen'].astype(int),
                'bplambda': bp_data['bplambda'].values
            })
            merged_bp = pd.merge(spd_df, bp_df, on='wavelen', how='inner')
            if len(merged_bp) > 0:
                kp_val = bp_data['kavD65'].iloc[0]
                bp = np.sum(merged_bp['power'].values * merged_bp['bplambda'].values) / kp_val
        
        if be_data is not None and len(be_data) > 0:
            be_df = pd.DataFrame({
                'wavelen': be_data['wavelen'].astype(int),
                'belambda': be_data['belambda'].values
            })
            merged_be = pd.merge(spd_df, be_df, on='wavelen', how='inner')
            if len(merged_be) > 0:
                ke_val = be_data['kavD65'].iloc[0]
                be = np.sum(merged_be['power'].values * merged_be['belambda'].values) / ke_val
            
    elif quantity == "radiant":
        # Apply Ba(λ), Bp(λ), Be(λ) without normalization
        ba = 0.0
        bp = 0.0
        be = 0.0
        
        if ba_data is not None and len(ba_data) > 0:
            ba_df = pd.DataFrame({
                'wavelen': ba_data['wavelen'].astype(int),
                'balambda': ba_data['balambda'].values
            })
            merged_ba = pd.merge(spd_df, ba_df, on='wavelen', how='inner')
            if len(merged_ba) > 0:
                ba = np.sum(merged_ba['power'].values * merged_ba['balambda'].values)
        
        if bp_data is not None and len(bp_data) > 0:
            bp_df = pd.DataFrame({
                'wavelen': bp_data['wavelen'].astype(int),
                'bplambda': bp_data['bplambda'].values
            })
            merged_bp = pd.merge(spd_df, bp_df, on='wavelen', how='inner')
            if len(merged_bp) > 0:
                bp = np.sum(merged_bp['power'].values * merged_bp['bplambda'].values)
        
        if be_data is not None and len(be_data) > 0:
            be_df = pd.DataFrame({
                'wavelen': be_data['wavelen'].astype(int),
                'belambda': be_data['belambda'].values
            })
            merged_be = pd.merge(spd_df, be_df, on='wavelen', how='inner')
            if len(merged_be) > 0:
                be = np.sum(merged_be['power'].values * merged_be['belambda'].values)
        
    elif quantity == "photon":
        # Apply spectral functions with 1/energy scaling
        ba = 0.0
        bp = 0.0
        be = 0.0
        
        if ba_data is not None and len(ba_data) > 0:
            ba_df = pd.DataFrame({
                'wavelen': ba_data['wavelen'].astype(int),
                'balambda': ba_data['balambda'].values
            })
            merged_ba = pd.merge(spd_df, ba_df, on='wavelen', how='inner')
            if len(merged_ba) > 0:
                e_ba = photonenergy(merged_ba['wavelen'].values)
                ba = np.log10(np.sum(merged_ba['power'].values * merged_ba['balambda'].values / (e_ba * 10000)) + 1e-20)
        
        if bp_data is not None and len(bp_data) > 0:
            bp_df = pd.DataFrame({
                'wavelen': bp_data['wavelen'].astype(int),
                'bplambda': bp_data['bplambda'].values
            })
            merged_bp = pd.merge(spd_df, bp_df, on='wavelen', how='inner')
            if len(merged_bp) > 0:
                e_bp = photonenergy(merged_bp['wavelen'].values)
                bp = np.log10(np.sum(merged_bp['power'].values * merged_bp['bplambda'].values / (e_bp * 10000)) + 1e-20)
        
        if be_data is not None and len(be_data) > 0:
            be_df = pd.DataFrame({
                'wavelen': be_data['wavelen'].astype(int),
                'belambda': be_data['belambda'].values
            })
            merged_be = pd.merge(spd_df, be_df, on='wavelen', how='inner')
            if len(merged_be) > 0:
                e_be = photonenergy(merged_be['wavelen'].values)
                be = np.log10(np.sum(merged_be['power'].values * merged_be['belambda'].values / (e_be * 10000)) + 1e-20)
    else:
        raise ValueError("Quantity must be 'luminous', 'radiant', or 'photon'")
    
    return {'Ba': float(ba), 'Bp': float(bp), 'Be': float(be)}


def govardovskii(lmax, wavelength):
    """
    Govardovskii et al. (2000) visual pigment template.
    
    Computes the universal visual pigment template for a given lambda_max.
    
    Reference: Govardovskii et al. (2000) Vis Neurosci. 17(4):509-528
    
    Args:
        lmax: Peak wavelength (lambda_max) in nm
        wavelength: Array of wavelengths in nm
    
    Returns:
        Array of normalized photon sensitivities (0-1)
    """
    wavelength = np.asarray(wavelength, dtype=float)
    lmax = float(lmax)
    
    # Govardovskii template parameters
    bband = 189 + 0.315 * lmax
    fmaxa = 3e17 / lmax
    fmaxb = 3e17 / bband
    Aa = 69.7
    Ab = 0.26
    a = 0.8795 + 0.0459 * np.exp(-((lmax - 300) ** 2) / 11940)
    B = 28
    b = 0.922
    C = -14.9
    c = 1.104
    D = 0.674
    d = -40.5 + 0.195 * lmax
    
    # Convert wavelengths to frequency
    freq = 3e17 / wavelength
    ffmax = freq / fmaxa
    
    # Alpha band
    Sa = 1 / (np.exp(Aa * (a - ffmax)) +
              np.exp(B * (b - ffmax)) +
              np.exp(C * (c - ffmax)) +
              D)
    
    # Beta band
    Sb = Ab * np.exp(-((wavelength - bband) / d) ** 2)
    
    # Total sensitivity
    aspecp = Sa + Sb
    
    # Normalize to max = 1
    aspecp = aspecp / np.max(aspecp)
    
    return aspecp

def harmonisesens(sens, target_range):
    """Adjust opsin sensitivity curve to match target wavelength range."""
    # 1. Extract Wavelengths and Values based on the data type
    if isinstance(sens, pd.DataFrame):
        # Case: Pandas DataFrame from _load_csv [1]
        wls = sens['wavelen'].values.astype(float)
        vals = sens['sens'].values.astype(float)
    elif isinstance(sens, dict):
        # Use 'aspecp' as seen in sensdict.png [1]
        wls = np.asarray(sens['wavelen'], dtype=float)
        
        # This handles both the 'sens' key (from DataFrames) 
        # and the 'aspecp' key (from your dictionary) [1]
        if 'aspecp' in sens:
            vals = np.asarray(sens['aspecp'], dtype=float)
        else:
            vals = np.asarray(sens['sens'], dtype=float)
    elif isinstance(sens, np.ndarray):
        # Case: NumPy array - Wavelength at index 5, Sensitivity at index 6 [4, 5]
        wls = sens[:, 5].astype(float)
        vals = sens[:, 6].astype(float)
    else:
        # Fallback for simple 1D arrays of just sensitivity values
        # If sens is just the Y-values, we assume it matches a standard 1nm range
        return interpolate.interp1d(np.arange(len(sens)), sens, kind='cubic', 
                                    bounds_error=False, fill_value=0)(target_range)

    # 2. Perform cubic spline interpolation (matches R behavior) [3, 6]
    f = interpolate.interp1d(wls, vals, kind='cubic', bounds_error=False, fill_value=0)
    
    return f(target_range)

def harmonisetrans(pfilter, target_range):
    """
    Adjust transmission data to match target wavelength range.
    Matches R exactly. Extrapolates properly.
    
    Args:
        pfilter: Dictionary with 'wavelen' and 'trans' keys
        target_range: Target wavelength range array
        
    Returns:
        Array of transmission values (%) aligned to target_range
    """
    # Extract existing data
    wlp = np.asarray(pfilter['wavelen'], dtype=float)
    trans = np.asarray(pfilter['trans'], dtype=float)
    
    # Short wavelength adjustment
    delta_short = int(wlp[0] - target_range[0])
    if delta_short > 0:
        # Extrapolate shorter wavelengths
        avg_delta = np.mean(np.diff(trans[:3]))
        extra = trans[0] - avg_delta * np.arange(delta_short, 0, -1)
        extra[extra < 0] = 0
        trans = np.concatenate([extra, trans])
        wlp = np.arange(target_range[0], wlp[-1] + 1)
    elif delta_short < 0:
        # Trim initial wavelengths
        trim_idx = -delta_short
        trans = trans[trim_idx:]
        wlp = wlp[trim_idx:]
    
    # Long wavelength adjustment
    delta_long = int(target_range[-1] - wlp[-1])
    if delta_long > 0:
        # Extrapolate longer wavelengths (assume constant transmission)
        trans = np.concatenate([trans, np.full(delta_long, trans[-1])])
        wlp = np.arange(wlp[0], target_range[-1] + 1)
    elif delta_long < 0:
        # Trim excess wavelengths
        trim_idx = len(trans) + delta_long
        trans = trans[:trim_idx]
        wlp = wlp[:trim_idx]
    
    return trans


def aopicspecies():
    """
    Get species information table.
    
    Returns:
        DataFrame with 64 species and their taxonomy/availability
    """
    species_data = _load_csv('SpeciesListData.csv')
    
    if species_data is not None and len(species_data) > 0:
        # Drop row index column if present
        if 'Unnamed: 0' in species_data.columns:
            species_data = species_data.drop(columns=['Unnamed: 0'])
        return species_data.reset_index(drop=True)
    
    return pd.DataFrame()


def generateaopicactionspec(opsin="Mel", lmax=None, pfilter=None, range_wl=None):
    """
    Generate alpha-opic action spectrum for a given opsin.
    EXACT translation from R. Matches R behavior completely.
    
    Args:
        opsin: "Mel", "Rod", "Scone", "Mcone", or "Lcone" (default "Mel")
        lmax: Lambda_max (numeric) or species name (string). If None, loads default from data file.
        pfilter: 0, species name (string), or DataFrame. If None, defaults to 0 (no filtering).
        range_wl: Wavelength range (default 300:800)
    
    Returns:
        Dictionary with 'opsin', 'wavelen', 'trans', 'aspecp', 'aspec', 'kavD65'
    """
    if range_wl is None:
        range_wl = np.arange(300, 801, dtype=float)
    else:
        range_wl = np.asarray(range_wl, dtype=float)
    
    # ========== Handle default lmax from data file ==========
    if lmax is None:
        # Load default lmax from first species in SensRefData
        sens_df = _load_csv(f'SensRefData_{opsin}.csv')
        if sens_df is not None and len(sens_df) > 0:
            # Get first row with valid lmax
            valid_lmax = sens_df[sens_df['lmax'].notna()]['lmax']
            if len(valid_lmax) > 0:
                lmax = float(valid_lmax.iloc[0])
            else:
                lmax = 480  # fallback
        else:
            lmax = 480  # fallback
    
    # ========== Handle default pfilter ==========
    if pfilter is None:
        pfilter = 0  # Default to no filtering
    
    # ========== Get photon-based spectral sensitivity ==========
    if isinstance(lmax, (int, float)) and lmax > 0:
        # Lambda max model (Govardovskii nomogram)
        aspecp = govardovskii(lmax, range_wl)
    elif isinstance(lmax, str):
        # Predefined sensitivity from SensRefData
        sens_df = _load_csv(f'SensRefData_{opsin}.csv')
        if sens_df is not None and len(sens_df) > 0:
            # Filter to this species
            species_data = sens_df[sens_df['species'].str.lower() == lmax.lower()]
            if len(species_data) > 0:
                # Extract wavelengths and sensitivity
                sens_dict = {
                    'wavelen': species_data['wavelen'].values,
                    'aspecp': species_data['aspecp'].values
                }
                aspecp = harmonisesens(sens_dict, range_wl)
            else:
                # Species not found - fallback
                aspecp = govardovskii(480, range_wl)
        else:
            aspecp = govardovskii(480, range_wl)
    else:
        # No valid lmax - should not reach here in normal use
        aspecp = govardovskii(480, range_wl)
    
    # ========== Get prereceptoral transmission ==========
    if isinstance(pfilter, (int, float)) and len(np.atleast_1d(pfilter)) == 1:
        # Numeric pfilter
        if pfilter == 0:
            trans = np.ones(len(range_wl)) * 100
        else:
            # Any other numeric value treated as 100% transmission
            trans = np.ones(len(range_wl)) * 100
    elif isinstance(pfilter, str):
        # Predefined transmission from TransRefData
        trans_df = _load_csv('TransRefData.csv')
        if trans_df is not None and len(trans_df) > 0:
            # Filter to this species
            species_trans = trans_df[trans_df['species'].str.lower() == pfilter.lower()]
            if len(species_trans) > 0:
                # Extract wavelengths and transmission
                trans_dict = {
                    'wavelen': species_trans['wavelen'].values,
                    'trans': species_trans['trans'].values
                }
                trans = harmonisetrans(trans_dict, range_wl)
            else:
                trans = np.ones(len(range_wl)) * 100
        else:
            trans = np.ones(len(range_wl)) * 100
    elif isinstance(pfilter, pd.DataFrame):
        # User-provided transmission data
        trans_dict = {
            'wavelen': pfilter.iloc[:, 0].values,
            'trans': pfilter.iloc[:, 1].values
        }
        trans = harmonisetrans(trans_dict, range_wl)
    else:
        # Default - no filtering
        trans = np.ones(len(range_wl)) * 100
    
    # ========== Apply prereceptoral correction ==========
    # Check if predefined sensitivity already has lens filter applied
    
    lens_already_applied = False
    if isinstance(lmax, str) and lmax == 'Human':
        # Sources state Human action spectra are already corrected for lens filtering [4]
        lens_already_applied = True

    if lens_already_applied:
        # Use the harmonised sensitivity directly
        aspecp_filtered = aspecp
    else:
        # For other species, also harmonise the lens transmission data to match range_wl
        # This ensures 'trans' also has a shape of (351,)
        trans = harmonisetrans(pfilter, range_wl) [5]
        aspecp_filtered = aspecp * (trans / 100.0) 


    # if isinstance(lmax, str) and lmax == 'Human':
    #     lens_filter = "yes"  # Human spectra are pre-corrected
    #     # Predefined sensitivity already has lens filter - don't apply again
    #     aspecp_filtered = aspecp_corrected
    # else:
    #     # Apply lens transmission to photon sensitivity
    #     aspecp_filtered = aspecp_corrected * (trans / 100.0)
    
    # ========== Convert to energy-based spectrum ==========
    e = photonenergy(range_wl)  # photon energy in Joules
    aspec = aspecp_filtered / e
    
    # ========== Normalize both spectra ==========
    max_aspec = np.nanmax(aspec) if not np.all(np.isnan(aspec)) else 1.0
    max_aspecp = np.nanmax(aspecp_filtered) if not np.all(np.isnan(aspecp_filtered)) else 1.0
    
    if max_aspec > 0:
        aspec = aspec / max_aspec
    if max_aspecp > 0:
        aspecp_filtered = aspecp_filtered / max_aspecp
    
    # Replace NaN with 0 for output (matches R behavior)
    aspec = np.nan_to_num(aspec, nan=0.0)
    aspecp_filtered = np.nan_to_num(aspecp_filtered, nan=0.0)
    
    # ========== Calculate a-opic efficacy of D65 daylight ==========
    d65_data = _load_csv('VisualStandards_D65.csv')
    if d65_data is not None and len(d65_data) > 0:
        # Get D65 wavelengths and spectra within range
        d65_wl = d65_data['wavelen'].values
        d65_spectra = d65_data['spectra'].values
        d65_lux = d65_data['lux'].iloc[0]
        
        # Filter to wavelengths in range
        d65_mask = (d65_wl >= range_wl[0]) & (d65_wl <= range_wl[-1])
        if np.any(d65_mask):
            d65_sp = d65_spectra[d65_mask]
            # Get corresponding aspec values
            aspec_mask = (range_wl >= d65_wl[d65_mask].min()) & (range_wl <= d65_wl[d65_mask].max())
            if np.any(aspec_mask):
                aspec_overlap = aspec[aspec_mask]
                # Match arrays and compute kavD65
                if len(d65_sp) == len(aspec_overlap):
                    kavD65 = np.sum(d65_sp * aspec_overlap) / d65_lux
                else:
                    kavD65 = 1.0
            else:
                kavD65 = 1.0
        else:
            kavD65 = 1.0
    else:
        kavD65 = 1.0
    
    # Ensure kavD65 is positive
    if kavD65 <= 0:
        kavD65 = 1.0
    
    return {
        'opsin': opsin,
        'wavelen': range_wl,
        'trans': trans,
        'aspecp': aspecp_filtered,
        'aspec': aspec,
        'kavD65': kavD65
    }


def alphaopic(power, wavelength, opsin="Mel", lmax=None, pfilter=None):
    """
    Main alpha-opic calculator. EXACT translation from R.
    
    Calculates luminous, radiant, or photon quantities using species-specific
    alpha-opic action spectra.
    
    Args:
        power: Spectral irradiance in W/m²·nm (array)
        wavelength: Array of wavelengths in nm
        opsin: "Mel", "Rod", "Scone", "Mcone", or "Lcone" (default: "Mel")
        lmax: Lambda_max (numeric) or species name. If None, uses default from data file.
        pfilter: 0, species name, or DataFrame. If None, defaults to 0 (no filtering).
    
    Returns:
        Dictionary with 'Luminous', 'Radiant', 'Photon' keys
    """
    power = np.asarray(power, dtype=float)
    wavelength = np.asarray(wavelength, dtype=float)
    
    # Validate inputs
    if len(power) != len(wavelength):
        raise ValueError("power and wavelength must have same length")
    
    # ========== Ensure wl and power are consistent (R: mean(diff(wl)) != 1) ==========
    wl = wavelength.copy()
    wl_spacing = np.mean(np.diff(wl))
    
    if wl_spacing != 1.0:
        # Not 1 nm spacing - use spline interpolation
        wlo = wl
        wl = np.arange(np.round(wlo[0]), np.round(wlo[-1]) + 1, dtype=float)
        # Use scipy spline - equivalent to R's spline(wlo, power, n=length(wl), method="natural")
        f = interpolate.CubicSpline(wlo, power, bc_type='natural')
        power = f(wl)
        power = power / np.mean(np.diff(wlo))  # preserve scale
    
    # ========== Check opsin and transmission info (R: message() calls) ==========
    # These are just informational in R - we can skip or log them
    
    # ========== Get action spectra and constants ==========
    aopics = generateaopicactionspec(opsin, lmax, pfilter, wl)
    crv = aopics['aspec']       # energy-based action spectrum
    crvp = aopics['aspecp']     # photon-based action spectrum
    const = aopics['kavD65']    # a-opic efficacy for D65
    wavelen = aopics['wavelen']
    
    # ========== Calculate a-opic quantities ==========
    e = photonenergy(wavelen)   # photon energy in Joules
    
    # EDI or EDL in lux or cd/m² (luminous)
    Luminous = np.sum(power * crv) / const
    
    # a-opic irradiance W/m² or radiance W/m²·sr (radiant)
    Radiant = np.sum(power * crv)
    
    # photon irradiance log10(photons/cm²/s) or radiance (photon)
    Photon = np.log10(np.sum(power * crvp / (e * 10000)))
    
    # Return results (as dict, matching Python interface)
    return {
        'Luminous': float(Luminous),
        'Radiant': float(Radiant),
        'Photon': float(Photon)
    }
